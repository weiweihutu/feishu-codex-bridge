```ts
import { appendFileSync, mkdirSync, writeFileSync } from 'node:fs';
import { join, relative } from 'node:path';
import { paths } from '../config/paths';
import { currentLogContext, log } from './logger';

const DEFAULT_TEXT_LIMIT = 20_000;

export interface AuditMessage {
  messageId: string;
  chatId: string;
  threadId?: string;
  senderId?: string;
  chatType: string;
  mentionedBot: boolean;
  createTime?: number;
  content?: string;
}

export type AuditContext = Record<string, unknown> & {
  msgId: string;
  chatId: string;
  threadId: string | null;
  senderId: string | null;
};

export interface TraceIo {
  workspaceRoot?: string;
  now?: () => Date;
}

export function truncateAuditText(
  value: unknown,
  limit = DEFAULT_TEXT_LIMIT,
): { text: string; truncated: boolean } {
  const text = typeof value === 'string' ? value : value == null ? '' : String(value);
  return text.length <= limit
    ? { text, truncated: false }
    : { text: text.slice(0, limit), truncated: true };
}

export function buildAuditContext(
  msg: AuditMessage,
  text: unknown,
  extras: Record<string, unknown> = {},
  limit = DEFAULT_TEXT_LIMIT,
): AuditContext {
  const message = truncateAuditText(text ?? msg.content ?? '', limit);
  return {
    msgId: msg.messageId,
    chatId: msg.chatId,
    threadId: msg.threadId ?? null,
    senderId: msg.senderId ?? null,
    chatType: msg.chatType,
    mentionedBot: msg.mentionedBot,
    messageText: message.text,
    messageTextTruncated: message.truncated,
    receivedAt: new Date(msg.createTime || Date.now()).toISOString(),
    ...extras,
  };
}

function workspaceRoot(io: TraceIo): string {
  return io.workspaceRoot ?? join(paths.appDir, 'my_workspace');
}

function dateKey(now: Date): string {
  return [
    now.getFullYear(),
    String(now.getMonth() + 1).padStart(2, '0'),
    String(now.getDate()).padStart(2, '0'),
  ].join('');
}

function safeName(value: unknown, fallback = 'unknown'): string {
  return String(value || fallback).replace(/[^a-zA-Z0-9_.-]+/g, '_').slice(0, 160) || fallback;
}
```

```ts
export function traceArtifactPath(
  msgId: unknown,
  name: unknown,
  content: unknown,
  kind: 'text' | 'json' = 'text',
  io: TraceIo = {},
): string | null {
  try {
    const now = (io.now ?? (() => new Date()))();
    const root = workspaceRoot(io);
    const dir = join(root, 'traces', 'artifacts', dateKey(now), safeName(msgId));
    mkdirSync(dir, { recursive: true });
    const file = join(dir, safeName(name));
    const body = kind === 'json' ? JSON.stringify(content ?? null, null, 2) : String(content ?? '');
    writeFileSync(file, body, 'utf8');
    return relative(root, file);
  } catch {
    return null;
  }
}

export function emitTraceStep(
  step: Record<string, unknown> = {},
  io: TraceIo = {},
): void {
  try {
    const now = (io.now ?? (() => new Date()))();
    const dir = join(workspaceRoot(io), 'traces', 'logs');
    mkdirSync(dir, { recursive: true });
    const payload = {
      ts: now.toISOString(),
      event: 'trace_step',
      status: 'success',
      ...step,
    } as Record<string, unknown>;
    if ('input_text' in payload) payload.input_text = truncateAuditText(payload.input_text).text;
    if ('output_text' in payload) payload.output_text = truncateAuditText(payload.output_text).text;
    appendFileSync(
      join(dir, `trace-${dateKey(now)}.log`),
      `${JSON.stringify(payload)}\n`,
      'utf8',
    );
  } catch {
    // Trace persistence is observational and must not fail a user turn.
  }
}
```

```ts
export interface FeishuContextIdentity {
  messageId: string;
  senderId?: string;
  chatId: string;
  threadId?: string;
}

export function appendFeishuContext(text: string, msg: FeishuContextIdentity): string {
  const context = [
    '[Feishu Context]',
    `message_id=${msg.messageId}`,
    `feishu_user_id=${msg.senderId ?? ''}`,
    `chat_id=${msg.chatId}`,
    `thread_id=${msg.threadId ?? ''}`,
    '',
    'When running business skills, set FEISHU_USER_ID to feishu_user_id and REQUEST_ID to message_id.',
  ].join('\n');
  return `${text}\n\n${context}`;
}

export interface NoMentionProject {
  kind?: 'single' | 'multi';
  noMention?: boolean;
  defaultNoMention: boolean;
}

export interface NoMentionMessage {
  content: string;
  threadId?: string;
  mentionAll: boolean;
  mentions: Array<{ isBot?: boolean }>;
}

export function shouldRespondWithoutMention(
  project: NoMentionProject,
  msg: NoMentionMessage,
): boolean {
  if (!(project.noMention ?? project.defaultNoMention)) return false;
  if (msg.mentionAll || msg.mentions.some((mention) => !mention.isBot)) return false;
  return true;
}
```
